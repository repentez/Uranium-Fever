EUC_add_war_exhaustion = {
	add_to_variable = {
		var = EUC_war_exhaustion_var
		value = EUC_war_exhaustion_var_temp
	}
	clamp_variable = {
		var = EUC_war_exhaustion_var
		min = 0
		max = 1
	}
	custom_effect_tooltip = EUC_add_war_exhaustion_tt
	hidden_effect = {
		if = {
			limit = {
				check_variable = { var = EUC_war_exhaustion_var value = 1 compare = greater_than_or_equals }
				NOT = { has_global_flag = EUC_UAR_nuclear_exchange }
			}
			EUC_nuclear_exchange = yes
		}
	}
}

	# EUC strikes: any theater state it doesn't control

	}

		add_political_power = -200
		add_stability = -0.5
		add_war_support = -0.3

		add_political_power = -200

	# End the war
	EUC = {

		}
	}

	# Reset the mechanic so nothing is left stuck at 100%
	EUC = {
		set_variable = { EUC_war_exhaustion_var = 0 }
		clr_country_flag = EUC_Theater_action
	}
	UAR = {
		set_variable = { UAR_war_exhaustion_var = 0 }
		clr_country_flag = UAR_Theater_action
	}

	every_country = {
		limit = { is_ai = no }
		country_event = EUC.1
	}
}
