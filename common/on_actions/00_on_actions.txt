on_actions = {
    on_startup = {
        effect = {
            every_country = {
                limit = {
                    is_ai = no
                    is_subject = no
                }

                country_event = {
                    id = uranium_fever.1
                }
            }
        }
    }
}
on_actions = {

    on_capitulation = {
        events = {}

        effect = {
            if = {
                limit = {
                    THIS = { tag = YUG }
                    YUG = { is_in_war_with = EUC }
                }

                EUC = {
                    white_peace = YUG
                }

                EUC = {
                    annex_country = {
                        target = YUG
                        transfer_troops = yes
                    }
                }

                set_country_flag = yug_absorbed_by_euc
            }

            if = {
                limit = {
                    THIS = { tag = ALB }
                    ALB = { is_in_war_with = EUC }
                }

                    white_peace = ALB
                }

                EUC = {
                    annex_country = {
                        target = ALB
                        transfer_troops = yes
                    }
                }

                set_country_flag = alb_absorbed_by_euc
            }

            if = {
                limit = {
                    THIS = { tag = GRE }
                    GRE = { is_in_war_with = EUC }
                }

                EUC = {
                    white_peace = GRE
                }

                EUC = {
                    annex_country = {
                        target = GRE
                        transfer_troops = yes
                    }
                }

                set_country_flag = gre_absorbed_by_euc
            }

            if = {
                limit = {
                    THIS = { tag = TUR }
                    TUR = { is_in_war_with = EUC }
                }

                    white_peace = TUR

                        target = TUR

                }

    }
